\section{Exponentielle de matrice}

\reminder{Pour $A \in \calset{M}_n(\R)$, $e^{tA} = \sum_{n \ge 0} \frac{t^n}{n!} A^n$.}

\begin{exercise}[label=ex:expo]
    Calculer $e^{tA}$ pour chacune des matrices suivantes.
    \begin{question}
        $A = \begin{pmatrix} 0 & 1 \\ -1 & 0 \end{pmatrix}$, en remarquant que
        $A^2 = -I$.
    \end{question}
    \begin{question}
        $A = \begin{pmatrix} 0 & 1 \\ 0 & 0 \end{pmatrix}$, nilpotente.
        \begin{subquestion}
            Calculer $A^2$.
        \end{subquestion}
        \begin{subquestion}
            En déduire $e^{tA}$.
        \end{subquestion}
    \end{question}
\solution
    \begin{question}
        Des identités $A^{2k} = (-1)^k I$ et $A^{2k+1} = (-1)^k A$ on tire
        \[
            e^{tA} = \begin{pmatrix} \cos t & \sin t \\ -\sin t & \cos t\end{pmatrix}.
        \]
    \end{question}
    \begin{question}
        $A^2 = 0$, donc $e^{tA} = I + tA$.
    \end{question}
\end{exercise}

\section{Systèmes non homogènes}

\begin{exercise}[nosolution]
    Soit $A \in \Cclass{0}(\TimeInterval, \calset{M}_n(\R))$ et $b \in \Cclass{0}(\TimeInterval, \R^n)$.
    \begin{question}
        Écrire la formule de variation de la constante pour
        $\dot{x}(t) = A(t)x(t) + b(t)$.
    \end{question}
\end{exercise}

\supplement{Pour aller plus loin, traiter le cas où $A$ n'est pas continue mais
seulement localement intégrable.}
